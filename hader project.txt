#include <iostream>
using namespace std;

struct Student
{
    int rollNo;
    string name;
    float marks;
    char grade;
};

Student s[100];
int total = 0;

// Function to calculate grade
char calculateGrade(float marks)
{
    if (marks >= 80)
        return 'A';
    else if (marks >= 60)
        return 'B';
    else if (marks >= 40)
        return 'C';
    else
        return 'F';
}

// Add Student
void addStudent()
{
    cout << "\nEnter Roll Number: ";
    cin >> s[total].rollNo;

    cin.ignore();

    cout << "Enter Name: ";
    getline(cin, s[total].name);

    cout << "Enter Marks: ";
    cin >> s[total].marks;

    s[total].grade = calculateGrade(s[total].marks);

    total++;

    cout << "\nStudent Added Successfully!\n";
}

// Display Students
void displayStudents()
{
    if (total == 0)
    {
        cout << "\nNo student record found!\n";
        return;
    }

    cout << "\n--- Student Records ---\n";

    for (int i = 0; i < total; i++)
    {
        cout << "\nStudent " << i + 1 << endl;

        cout << "Roll Number : " << s[i].rollNo << endl;
        cout << "Name        : " << s[i].name << endl;
        cout << "Marks       : " << s[i].marks << endl;
        cout << "Grade       : " << s[i].grade << endl;
    }
}

// Search Student
void searchStudent()
{
    int roll;
    bool found = false;

    cout << "\nEnter Roll Number to Search: ";
    cin >> roll;

    for (int i = 0; i < total; i++)
    {
        if (s[i].rollNo == roll)
        {
            cout << "\nStudent Found!\n";

            cout << "Roll Number : " << s[i].rollNo << endl;
            cout << "Name        : " << s[i].name << endl;
            cout << "Marks       : " << s[i].marks << endl;
            cout << "Grade       : " << s[i].grade << endl;

            found = true;
        }
    }

    if (!found)
    {
        cout << "\nStudent Not Found!\n";
    }
}

// Update Student
void updateStudent()
{
    int roll;
    bool found = false;

    cout << "\nEnter Roll Number to Update: ";
    cin >> roll;

    for (int i = 0; i < total; i++)
    {
        if (s[i].rollNo == roll)
        {
            cin.ignore();

            cout << "Enter New Name: ";
            getline(cin, s[i].name);

            cout << "Enter New Marks: ";
            cin >> s[i].marks;

            s[i].grade = calculateGrade(s[i].marks);

            cout << "\nStudent Updated Successfully!\n";

            found = true;
        }
    }

    if (!found)
    {
        cout << "\nStudent Not Found!\n";
    }
}

// Delete Student
void deleteStudent()
{
    int roll;
    bool found = false;

    cout << "\nEnter Roll Number to Delete: ";
    cin >> roll;

    for (int i = 0; i < total; i++)
    {
        if (s[i].rollNo == roll)
        {
            for (int j = i; j < total - 1; j++)
            {
                s[j] = s[j + 1];
            }

            total--;

            cout << "\nStudent Deleted Successfully!\n";

            found = true;
            break;
        }
    }

    if (!found)
    {
        cout << "\nStudent Not Found!\n";
    }
}

// Main Function
int main()
{
    int choice;

    do
    {
        cout << "\n===== Student Management System =====\n";

        cout << "1. Add Student\n";
        cout << "2. Display Students\n";
        cout << "3. Search Student\n";
        cout << "4. Update Student\n";
        cout << "5. Delete Student\n";
        cout << "6. Exit\n";

        cout << "\nEnter Your Choice: ";
        cin >> choice;

        switch (choice)
        {
            case 1:
                addStudent();
                break;

            case 2:
                displayStudents();
                break;

            case 3:
                searchStudent();
                break;

            case 4:
                updateStudent();
                break;

            case 5:
                deleteStudent();
                break;

            case 6:
                cout << "\nProgram Ended.\n";
                break;

            default:
                cout << "\nInvalid Choice!\n";
        }

    } while (choice != 6);

    return 0;
}
